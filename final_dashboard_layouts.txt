# NIFTY OI CHART
# =========================================================

class NiftyOIChart(QWidget):

    def __init__(
        self,
        parent: Optional[QWidget] = None,
    ):
        super().__init__(parent)

        self.setMinimumHeight(
            210
        )

        self.call_values = [
            11,
            14,
            18,
            13,
            9,
            16,
            21,
            17,
            12,
            20,
            15,
            19,
        ]

        self.put_values = [
            8,
            11,
            14,
            17,
            19,
            13,
            16,
            22,
            18,
            15,
            21,
            23,
        ]

        self.price_values = [
            42,
            45,
            49,
            53,
            57,
            59,
            61,
            60,
            63,
            68,
            73,
            80,
        ]

    def paintEvent(
        self,
        event,
    ) -> None:

        painter = QPainter(self)

        painter.setRenderHint(
            QPainter.Antialiasing
        )

        area = self.rect().adjusted(
            35,
            18,
            -18,
            -28,
        )

        count = len(
            self.call_values
        )

        if count == 0:
            return

        group_width = (
            area.width()
            / count
        )

        max_oi = max(
            max(self.call_values),
            max(self.put_values),
            1,
        )

        # horizontal grid
        grid_pen = QPen(
            QColor("#17304a")
        )

        grid_pen.setWidth(1)

        painter.setPen(
            grid_pen
        )

        for i in range(4):

            y = (
                area.top()
                + (
                    area.height()
                    * i
                    / 3
                )
            )

            painter.drawLine(
                QPointF(
                    area.left(),
                    y,
                ),
                QPointF(
                    area.right(),
                    y,
                ),
            )

        # bars
        for i in range(count):

            base_x = (
                area.left()
                + i
                * group_width
            )

            call_h = (
                area.height()
                * 0.55
                * self.call_values[i]
                / max_oi
            )

            put_h = (
                area.height()
                * 0.55
                * self.put_values[i]
                / max_oi
            )

            bar_w = max(
                4,
                group_width * 0.22,
            )

            painter.setPen(
                Qt.NoPen
            )

            painter.setBrush(
                QColor(GREEN)
            )

            painter.drawRect(
                QRectF(
                    base_x
                    + group_width * 0.20,
                    area.bottom()
                    - call_h,
                    bar_w,
                    call_h,
                )
            )

            painter.setBrush(
                QColor(RED)
            )

            painter.drawRect(
                QRectF(
                    base_x
                    + group_width * 0.52,
                    area.bottom()
                    - put_h,
                    bar_w,
                    put_h,
                )
            )

        # Price line
        minimum = min(
            self.price_values
        )

        maximum = max(
            self.price_values
        )

        spread = max(
            maximum - minimum,
            1,
        )

        path = QPainterPath()

        for i, value in enumerate(
            self.price_values
        ):

            x = (
                area.left()
                + group_width
                * (
                    i + 0.5
                )
            )

            normalized = (
                value - minimum
            ) / spread

            y = (
                area.top()
                + area.height()
                * 0.45
                * (
                    1 - normalized
                )
            )

            point = QPointF(
                x,
                y,
            )

            if i == 0:
                path.moveTo(
                    point
                )
            else:
                path.lineTo(
                    point
                )

        price_pen = QPen(
            QColor(BLUE)
        )

        price_pen.setWidth(3)

        painter.setPen(
            price_pen
        )

        painter.drawPath(
            path
        )
