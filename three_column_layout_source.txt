                    1 - normalized
                )
            )

            point = QPointF(
                x,
                y,
            )

            if i == 0:
                path.moveTo(
                    point
                )
            else:
                path.lineTo(
                    point
                )

        price_pen = QPen(
            QColor(BLUE)
        )

        price_pen.setWidth(3)

        painter.setPen(
            price_pen
        )

        painter.drawPath(
            path
        )
