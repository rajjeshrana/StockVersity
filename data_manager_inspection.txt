    # =====================================================
    # STORE DATAFRAME
    # =====================================================

    def set_data(

        self,

        instrument_key: str,

        timeframe: str,

        dataframe: pd.DataFrame,

    ) -> None:
        """
        Store OHLCV dataframe.
        """

        with self.lock:

            self.data[

                instrument_key

            ][

                timeframe

            ] = dataframe.copy()

            self.last_refresh[

                f"{instrument_key}:{timeframe}"

            ] = datetime.now()

    # =====================================================
    # GET DATAFRAME
    # =====================================================

    def get_data(

        self,

        instrument_key: str,

        timeframe: str,

    ) -> Optional[pd.DataFrame]:
        """
        Return cached dataframe.
        """

        with self.lock:

            timeframe_map = self.data.get(

                instrument_key

            )

            if timeframe_map is None:

                self.cache_misses += 1

                return None

            dataframe = timeframe_map.get(

                timeframe

            )

            if dataframe is None:

                self.cache_misses += 1

                return None

            self.cache_hits += 1

            return dataframe.copy()

    # =====================================================
    # HAS DATA
    # =====================================================

    def has_data(

        self,

        instrument_key: str,

        timeframe: str,

    ) -> bool:

        return (

            instrument_key in self.data

            and timeframe in self.data[

                instrument_key

            ]

        )

    # =====================================================
    # REMOVE DATA

    ) -> None:

def _safe_float(
    value: Any,
    default: float = 0.0,
) -> float:
    """
    Convert a value safely to float.
    """

    try:

        if value is None:
            return default

        return float(value)

    except (
        TypeError,
        ValueError,
    ):

        return default


def _safe_int(
    value: Any,
    default: int = 0,
) -> int:
    """
    Convert a value safely to int.
    """

    try:

        if value is None:
            return default

        return int(float(value))

    except (
        TypeError,
        ValueError,
    ):

        return default


def _first_value(
    data: Mapping[str, Any],
    *keys: str,
    default: Any = None,
) -> Any:
    """
    Return first available non-None value.
    """

    for key in keys:

        if key in data:

            value = data[key]

            if value is not None:
                return value

    return default


def _normalize_symbol(
    symbol: Any,
) -> str:
    """
    Normalize trading symbol.
    """

    if symbol is None:
        return ""

    return str(
        symbol
    ).strip().upper()


def _normalize_instrument_key(
    instrument_key: Any,
) -> str:
    """
    Normalize Upstox instrument key.
    """

    if instrument_key is None:
        return ""

    return str(
        instrument_key
    ).strip()
