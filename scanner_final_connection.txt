    def run_market_cycle(self):

        self.scan_counter += 1

        self.last_scan_time = datetime.now()

        self.statusChanged.emit(
            f"Running Scan #{self.scan_counter}"
        )

        start = time.perf_counter()

        self.fetch_market_data()

        self.calculate_indicators()

        opportunities = self.execute_scanner()

        ranked = self.rank_opportunities(
            opportunities
        )

        trades = self.generate_trade_cards(
            ranked
        )

        self.best_opportunities = trades

        self.dashboard_data = {

            "scan_number": self.scan_counter,

            "scan_time": self.last_scan_time,

            "market_open": self.market_open,

            "opportunities": trades

        }

        self.dashboardUpdated.emit(
            self.dashboard_data
        )

        self.scannerCompleted.emit(
            trades
        )

        runtime = round(

            time.perf_counter() - start,

            2

        )

        self.statusChanged.emit(

            f"Completed in {runtime} sec"

        )

class DependencyContainer:

    """
    Singleton Dependency Container

    One instance of every engine
    throughout the application.
    """

    def __init__(self):

        self.logger = logging.getLogger(
            "DependencyContainer"
        )

        self._instances: Dict[str, Any] = {}

        self._build()

    # ===================================================
    # BUILD CONTAINER
    # ===================================================

    def _build(self):

        self.logger.info(
            "Creating Dependency Container..."
        )

        self.register(
            "service_manager",
            ServiceManager()
        )

        service = self.get(
            "service_manager"
        )

        controller = service.application

        self.register(
            "application_controller",
            controller
        )

        self.register(
            "nse_engine",
            controller.nse_engine
        )

        self.register(
            "scanner_executor",
            controller.scanner
        )

        self.register(
            "market_prediction",
            controller.market_prediction
        )

        self.register(
            "market_sentiment",
            controller.market_sentiment
        )

        self.register(
            "market_regime",
            controller.market_regime
        )

        self.register(
            "pattern_engine",
            controller.pattern_engine
        )

        self.register(
            "opportunity_ranker",
            controller.ranker
        )

        self.register(
            "risk_engine",
            controller.risk_engine
        )

        self.register(
            "market_structure",
            controller.structure_engine
        )

        self.register(
            "sector_rotation",
            controller.rotation_engine
        )

        self.register(
            "volume_engine",
            controller.volume_engine
        )

        self.logger.info(
            "Dependency Container Ready."
        )

    # ===================================================
    # REGISTER
    # ===================================================

    def register(

        self,

        name: str,

        instance: object

    ):

        self._instances[name] = instance

    # ===================================================
    # GET
    # ===================================================

    def get(

        self,

        name: str

    ):

        if name not in self._instances:

            raise KeyError(

                f"{name} not registered."

            )

        return self._instances[name]

    # ===================================================
    # EXISTS
    # ===================================================

    def exists(

        self,

        name: str

    ):

        return name in self._instances

    # ===================================================
    # REMOVE
    # ===================================================

    def remove(

        self,

        name: str

    ):

        if self.exists(name):

            del self._instances[name]

    # ===================================================
    # REGISTER INSTANCE
    # ===================================================

    def replace(

        self,

        name,

        instance

    ):

        self._instances[name] = instance

    # ===================================================
    # REGISTER MULTIPLE
    # ===================================================

    def register_many(
        self,
        instances: Dict[str, object]
    ):

        for name, instance in instances.items():

            self.register(name, instance)

        return True


    # ===================================================
    # GET ALL
    # ===================================================

    def all_instances(self):

        return self._instances


    # ===================================================
    # GET ALL NAMES
    # ===================================================

    def names(self):

        return list(

        report = {}

            report[name] = instance is not None

        return report

    # READY
    # ===================================================

    def ready(self):

        return all(

            self.verify().values()

        )


    # ===================================================
    # HEALTH REPORT
    # ===================================================

    def health(self):

        report = {}

        for name, instance in self._instances.items():

            try:

                if hasattr(instance, "health"):

                    report[name] = instance.health()

                elif hasattr(instance, "health_status"):

                    report[name] = instance.health_status()

                else:

                    report[name] = {

                        "status": "OK"

                    }

            except Exception as e:

                report[name] = {

                    "status": "ERROR",

                    "message": str(e)

                }

        return report
